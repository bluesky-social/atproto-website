// =============================================================================
// Post Index
// =============================================================================

interface IndexedPost {
    uri: AtUriString
    indexedAt: number
}

const postIndex: IndexedPost[] = []
const jetstream = new Jetstream('https://jetstream.us-east.bsky.network')

function escapeRegExp(value: string) {
    return [...value].map(char => '.^$*+?()[]{}|\\'.includes(char) ? `\\${char}` : char).join('')
}

const searchPattern = new RegExp(
    `\\b(${config.searchTerms.map(escapeRegExp).join('|')})\\b`,
    'i'
)

async function indexJetstream(signal: AbortSignal) {
    for await (const evt of jetstream.live({
        collections: ['app.bsky.feed.post'],
        dids: config.initialRepos.length > 0 ? config.initialRepos : undefined,
        kinds: ['commit', 'account'],
        signal,
    })) {
        if (evt.kind === 'account') {
            if (evt.account.active || evt.account.status !== 'deleted') continue
            const removed = postIndex.filter(p => p.uri.startsWith(`at://${evt.did}/`)).length
            if (removed > 0) {
                postIndex.splice(0, postIndex.length, ...postIndex.filter(p => !p.uri.startsWith(`at://${evt.did}/`)))
                console.log(`Account ${evt.did} deleted: removed ${removed} posts`)
            }
            continue
        }

        if (evt.kind !== 'commit' || evt.commit.collection !== 'app.bsky.feed.post') continue

        const { rkey } = evt.commit
        const uri = `at://${evt.did}/app.bsky.feed.post/${rkey}` as AtUriString
        const idx = postIndex.findIndex(p => p.uri === uri)

        if (evt.commit.operation === 'delete') {
            if (idx !== -1) {
                postIndex.splice(idx, 1)
                console.log(`DELETE ${uri}`)
            }
            continue
        }

        const text = typeof evt.commit.record?.text === 'string' ? evt.commit.record.text : ''
        if (!searchPattern.test(text)) {
            if (idx !== -1) postIndex.splice(idx, 1)
            continue
        }

        if (idx !== -1) postIndex.splice(idx, 1)
        postIndex.unshift({ uri, indexedAt: Date.now() })
        if (postIndex.length > config.maxPosts) postIndex.pop()

        const preview = text.substring(0, 60).replace(/\n/g, ' ')
        console.log(`${evt.commit.operation.toUpperCase()} ${uri}`)
        console.log(`  \"${preview}${text.length > 60 ? '...' : ''}\"`)
        console.log(`  ⭐ Added to index (${postIndex.length} total)`)
    }
}
