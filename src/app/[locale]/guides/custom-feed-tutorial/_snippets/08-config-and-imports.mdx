import { Jetstream } from '@bsky/jetstream'
import { AtUriString, DidString, isDidString } from '@atproto/lex'
import { LexError, LexRouter, serviceAuth } from '@atproto/lex-server'
import { serve } from '@atproto/lex-server/nodejs'
import * as app from './lexicons/app.js'

// =============================================================================
// Configuration
// =============================================================================

interface FeedConfig {
    publisherDid: DidString
    feedName: string
    searchTerms: string[]
    maxPosts: number
    port: number
    initialRepos: DidString[]
}

const publisherDid = process.env.FEED_PUBLISHER_DID || 'did:example:alice'
if (!isDidString(publisherDid)) {
    throw new Error('FEED_PUBLISHER_DID must be a valid DID')
}

const config: FeedConfig = {
    publisherDid,
    feedName: process.env.FEED_NAME || 'goblin-mode',
    searchTerms: (process.env.FEED_SEARCH_TERMS || 'goblin,goblins,gremlin,gremlins,wizard,wizards,chaos').split(',').map(s => s.trim()),
    maxPosts: parseInt(process.env.FEED_MAX_POSTS || '1000', 10),
    port: parseInt(process.env.FEED_PORT || '3000', 10),
    initialRepos: (process.env.FEED_INITIAL_REPOS || '').split(',').map(s => s.trim()).filter(Boolean).map(s => s as DidString),
}

const FEED_URI: AtUriString = `at://${config.publisherDid}/app.bsky.feed.generator/${config.feedName}` as AtUriString
