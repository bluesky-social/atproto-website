import { SpaceSyncIndexer } from '@bsky/space-sync'

const indexer = new SpaceSyncIndexer(board)
  .commit(note, {
    put: (evt) => upsertNote(evt.space, evt.did, evt.rkey, evt.record.text),
    del: (evt) => deleteNote(evt.space, evt.did, evt.rkey),
  })
  .space({
    deleted: (evt) => deleteSpace(evt.space),
    authorRemoved: (evt) => deleteAuthorInSpace(evt.space, evt.did),
    authorReset: (evt) => deleteAuthorInSpace(evt.space, evt.did),
  })
  .account((evt) => setAuthorStatus(evt.did, evt.active, evt.status))

await stream.runner(indexer).replay({ cursor })
